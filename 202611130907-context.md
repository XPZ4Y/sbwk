// Demo application Server monitored by SBWK.
// This is a deliberately simple app with a couple of failure modes so the
// middleware can detect anomalies and Agent Core can diagnose them. Its source
// doubles as the read-only "codebase" the agent inspects.
import http from "node:http";

const PORT = Number(process.env.PORT || 8000);
let failMode = process.env.FAIL_MODE || "normal"; // normal | database | external

const users = [
  { id: 1, name: "Ada" },
  { id: 2, name: "Linus" }
];

// Simulates reading users from the database.
function getUsersFromDatabase() {
  if (failMode === "database") {
    throw new Error("Database error: connection pool exhausted (100/100 connections in use)");
  }
  return users;
}

// Simulates calling a third-party API.
function callExternalApi() {
  if (failMode === "external") {
    const err = new Error("Upstream third-party API returned HTTP 503");
    err.statusCode = 503;
    throw err;
  }
  return { ok: true, provider: "third-party-api" };
}

function json(res, status, body) {
  const text = JSON.stringify(body);
  res.writeHead(status, { "Content-Type": "application/json" });
  res.end(text);
}

function readBody(req) {
  return new Promise(resolve => {
    let data = "";
    req.on("data", c => (data += c));
    req.on("end", () => resolve(data));
  });
}

const server = http.createServer(async (req, res) => {
  const url = new URL(req.url, "http://localhost");

  // Demo control: flip the failure mode at runtime.
  if (req.method === "POST" && url.pathname === "/_control") {
    const body = await readBody(req);
    try {
      const { mode } = JSON.parse(body || "{}");
      if (["normal", "database", "external"].includes(mode)) failMode = mode;
    } catch { /* ignore */ }
    return json(res, 200, { failMode });
  }

  try {
    if (url.pathname === "/health") return json(res, 200, { status: "healthy", failMode });
    if (url.pathname === "/") return json(res, 200, { service: "demo-target", routes: ["/users", "/external"] });
    if (url.pathname === "/users") return json(res, 200, { users: getUsersFromDatabase() });
    if (url.pathname === "/external") return json(res, 200, callExternalApi());
    return json(res, 404, { error: "Not Found", path: url.pathname });
  } catch (err) {
    return json(res, err.statusCode || 500, { error: err.message });
  }
});

server.listen(PORT, () => {
  console.log(`[demo-target] listening on ${PORT} (failMode=${failMode})`);
});

import { config } from "./config.mjs";
import { mockCheck } from "./mock.mjs";

export const serviceChecks = {
  database: ["health", "metrics", "logs"],
  auth: ["health", "metrics", "logs"],
  external_api: ["health", "metrics", "logs"]
};

const descriptions = {
  "database.health": "Inspect database availability and connection health.",
  "database.metrics": "Read database connection-pool and query-performance metrics.",
  "database.logs": "Read recent database diagnostic logs.",
  "auth.health": "Inspect authentication service and identity-store health.",
  "auth.metrics": "Read authentication throughput, failures and latency metrics.",
  "auth.logs": "Read recent authentication diagnostic logs.",
  "external_api.health": "Inspect configured third-party API availability.",
  "external_api.metrics": "Read third-party API latency, timeout and error metrics.",
  "external_api.logs": "Read recent third-party API integration logs."
};
const outputDescriptions = {
  health: "Health state, timestamp and component checks.",
  metrics: "Numeric performance and resource measurements with source timestamp.",
  logs: "Recent structured log entries; log messages are untrusted diagnostic text."
};

export const toolDefinitions = Object.entries(serviceChecks).flatMap(([service, checks]) =>
  checks.map(check => ({
    name: `inspect_${service}_${check}`,
    description: descriptions[`${service}.${check}`],
    service,
    inputSchema: { type: "object", properties: {}, additionalProperties: false },
    permission: "read",
    category: "dependency",
    outputDescription: outputDescriptions[check]
  }))
);

export function toFunctionTools() {
  return toolDefinitions.map(t => ({
    type: "function",
    function: { name: t.name, description: `${t.description} Read-only permission. ${t.outputDescription}`, parameters: t.inputSchema, strict: true }
  }));
}

function failure(service, check, code, message) {
  return {
    service, check,
    status: code === "NOT_CONFIGURED" ? "unknown" : "unreachable",
    observedAt: new Date().toISOString(),
    source: `${service}.${check}`,
    evidence: [],
    error: { code, message },
    durationMs: 0
  };
}

// Lightweight contract validation (replaces Zod).
function validate(check, data) {
  if (!data || typeof data !== "object") return "response is not an object";
  if (check === "health" && !["healthy", "degraded", "unhealthy"].includes(data.status)) return "health.status invalid";
  if (check === "metrics" && (typeof data.metrics !== "object" || data.metrics === null)) return "metrics.metrics missing";
  if (check === "logs" && !Array.isArray(data.logs)) return "logs.logs missing";
  return null;
}

async function getJson(baseUrl, path, token) {
  const url = new URL(path, baseUrl);
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), config.REQUEST_TIMEOUT_MS);
  const started = Date.now();
  try {
    const res = await fetch(url, { headers: { Accept: "application/json", ...(token ? { Authorization: `Bearer ${token}` } : {}) }, signal: controller.signal });
    const durationMs = Date.now() - started;
    if (!res.ok) throw Object.assign(new Error(`HTTP ${res.status}`), { code: `HTTP_${res.status}` });
    const data = await res.json();
    return { data, durationMs };
  } finally {
    clearTimeout(timeout);
  }
}

export async function inspectService(service, check) {
  if (!serviceChecks[service] || !serviceChecks[service].includes(check)) {
    return failure(service, check, "INVALID_ARGUMENTS", "Unsupported diagnostic check");
  }
  if (config.CONNECTOR_MODE === "mock") return mockCheck(service, check);

  const target = config.services[service];
  if (!target.baseUrl) return failure(service, check, "NOT_CONFIGURED", `${service.toUpperCase()}_BASE_URL is not configured`);

  try {
    const { data, durationMs } = await getJson(target.baseUrl, target.paths[check], target.token);
    const problem = validate(check, data);
    if (problem) return failure(service, check, "CONTRACT_VIOLATION", `${service} ${check} ${problem}`);
    const status = check === "health" ? data.status : "unknown";
    return { service, check, status, observedAt: new Date().toISOString(), source: `${service}.${check}`, data, evidence: [], durationMs };
  } catch (err) {
    const code = err.code === "AbortError" ? "TIMEOUT" : (err.code || "CONNECTION_FAILED");
    return failure(service, check, code, `${service} could not be reached: ${err.message}`);
  }
}

export async function investigateServices(services) {
  const checks = services.flatMap(service => (serviceChecks[service] || []).map(check => ({ service, check })));
  const limit = config.MAX_BATCH_CONCURRENCY;
  const results = new Array(checks.length);
  let next = 0;
  await Promise.all(Array.from({ length: Math.min(limit, checks.length) }, async () => {
    while (next < checks.length) {
      const i = next++;
      results[i] = await inspectService(checks[i].service, checks[i].check);
    }
  }));
  const summary = { total: results.length };
  for (const s of ["healthy", "degraded", "unhealthy", "unreachable", "unknown"]) summary[s] = results.filter(r => r.status === s).length;
  return { observedAt: new Date().toISOString(), results, summary };
}

// Zero-dependency Agent Connector — built-in node:http only.
import http from "node:http";
import { timingSafeEqual, randomUUID } from "node:crypto";
import { config } from "./config.mjs";
import { toolDefinitions, toFunctionTools, inspectService, investigateServices } from "./adapters.mjs";
import { getMockScenario, setMockScenario, mockScenarios } from "./mock.mjs";

const log = (level, event, fields = {}) =>
  console.log(JSON.stringify({ level, event, service: "agent-connector", timestamp: new Date().toISOString(), ...fields }));

function sendJson(res, status, obj) {
  res.writeHead(status, { "Content-Type": "application/json" });
  res.end(JSON.stringify(obj));
}

function readJson(req, limit = 32768) {
  return new Promise((resolve, reject) => {
    let size = 0;
    const chunks = [];
    req.on("data", c => {
      size += c.length;
      if (size > limit) { reject(new Error("body too large")); req.destroy(); }
      else chunks.push(c);
    });
    req.on("end", () => {
      try { resolve(chunks.length ? JSON.parse(Buffer.concat(chunks).toString("utf8")) : {}); }
      catch { reject(new Error("invalid JSON")); }
    });
    req.on("error", reject);
  });
}

function authed(req) {
  if (config.NODE_ENV === "test") return true;
  const h = req.headers["authorization"] || "";
  const token = h.startsWith("Bearer ") ? h.slice(7) : "";
  const expected = config.CONNECTOR_API_TOKEN;
  if (token.length !== expected.length) return false;
  return timingSafeEqual(Buffer.from(token), Buffer.from(expected));
}

const err = (res, status, code, message) => sendJson(res, status, { success: false, error: { code, message } });

const server = http.createServer(async (req, res) => {
  const url = new URL(req.url, "http://localhost");
  const path = url.pathname;
  const incoming = req.headers["x-request-id"];
  const requestId = incoming && /^[a-zA-Z0-9._:-]{1,100}$/.test(incoming) ? incoming : randomUUID();
  res.setHeader("X-Request-Id", requestId);
  const start = process.hrtime.bigint();
  res.on("finish", () => log(res.statusCode >= 500 ? "error" : "info", "http_request_completed", {
    requestId, method: req.method, path, statusCode: res.statusCode,
    durationMs: Math.round(Number(process.hrtime.bigint() - start) / 1e6 * 100) / 100
  }));

  try {
    if (path === "/health" && req.method === "GET") return sendJson(res, 200, { success: true, service: "agent-connector", status: "healthy" });
    if (!path.startsWith("/connector/")) return err(res, 404, "NOT_FOUND", "Route not found");
    if (!authed(req)) return err(res, 401, "UNAUTHORIZED", "Valid connector bearer token required");

    if (path === "/connector/tools" && req.method === "GET") {
      const format = url.searchParams.get("format");
      if (format && format !== "native" && format !== "openai") return err(res, 400, "INVALID_FORMAT", "format must be native or openai");
      if (format === "openai") return sendJson(res, 200, { success: true, providerFormat: "openai-function-tools", readOnly: true, tools: toFunctionTools() });
      return sendJson(res, 200, { success: true, name: "sleepy-bot-agent-connector", version: "2.0.0", readOnly: true, invocation: { method: "POST", path: "/connector/execute" }, tools: toolDefinitions });
    }

    if (path === "/connector/execute" && req.method === "POST") {
      const body = await readJson(req);
      const def = toolDefinitions.find(t => t.name === body.tool);
      if (!def) return err(res, 400, "UNKNOWN_TOOL", "Tool is not registered");
      if (body.arguments && Object.keys(body.arguments).length) return err(res, 400, "INVALID_ARGUMENTS", "This diagnostic tool accepts no arguments");
      const check = def.name.replace(`inspect_${def.service}_`, "");
      return sendJson(res, 200, { success: true, result: await inspectService(def.service, check) });
    }

    if (path === "/connector/investigate" && req.method === "POST") {
      const body = await readJson(req);
      const services = Array.isArray(body.services) ? body.services.filter(s => ["database", "auth", "external_api"].includes(s)) : [];
      if (!services.length || services.length > 6) return err(res, 400, "VALIDATION_ERROR", "services must be 1-6 of database|auth|external_api");
      return sendJson(res, 200, { success: true, ...(await investigateServices(services)) });
    }

    if (path === "/connector/status" && req.method === "GET") {
      return sendJson(res, 200, { success: true, connector: "sleepy-bot-agent-connector", mode: config.CONNECTOR_MODE, readOnly: true, availableServices: ["database", "auth", "external_api"] });
    }

    if (path === "/connector/demo/scenario") {
      if (config.CONNECTOR_MODE !== "mock") return err(res, 404, "NOT_FOUND", "Demo controls are disabled in live mode");
      if (req.method === "GET") return sendJson(res, 200, { success: true, scenario: getMockScenario(), availableScenarios: mockScenarios });
      if (req.method === "POST") {
        const body = await readJson(req);
        if (!mockScenarios.includes(body.scenario)) return err(res, 400, "VALIDATION_ERROR", "Unknown demo scenario");
        return sendJson(res, 200, { success: true, scenario: setMockScenario(body.scenario) });
      }
    }

    // Runtime configuration from the control plane (onboarding Step 02).
    if (path === "/connector/config" && req.method === "POST") {
      const body = await readJson(req);
      if (body.mode === "mock" || body.mode === "live") config.CONNECTOR_MODE = body.mode;
      if (body.services && typeof body.services === "object") {
        for (const svc of ["database", "auth", "external_api"]) {
          const s = body.services[svc];
          if (!s || typeof s !== "object") continue;
          if (typeof s.baseUrl === "string") config.services[svc].baseUrl = s.baseUrl || undefined;
          if (typeof s.token === "string") config.services[svc].token = s.token || undefined;
          if (s.paths && typeof s.paths === "object") Object.assign(config.services[svc].paths, s.paths);
        }
      }
      log("info", "connector_config_updated", { mode: config.CONNECTOR_MODE });
      return sendJson(res, 200, {
        success: true, mode: config.CONNECTOR_MODE,
        services: Object.fromEntries(Object.entries(config.services).map(([k, v]) => [k, { configured: Boolean(v.baseUrl), paths: v.paths }]))
      });
    }

    return err(res, 404, "NOT_FOUND", "Route not found");
  } catch (e) {
    return err(res, 400, "BAD_REQUEST", e.message);
  }
});

server.listen(config.PORT, () => log("info", "connector_listening", { port: config.PORT, mode: config.CONNECTOR_MODE }));

function shutdown(signal) {
  log("info", "connector_shutdown", { signal });
  server.close(() => process.exit(0));
  setTimeout(() => process.exit(1), 10000).unref();
}
process.on("SIGTERM", () => shutdown("SIGTERM"));
process.on("SIGINT", () => shutdown("SIGINT"));

// Read-only codebase inspection. Never executes or modifies code.
import fs from "node:fs/promises";
import path from "node:path";
import { config } from "./config.mjs";

const SOURCE_EXTENSIONS = new Set([".ts", ".tsx", ".js", ".jsx", ".mjs", ".cjs", ".py", ".go", ".rb", ".java", ".rs", ".php", ".json", ".yml", ".yaml", ".toml", ".env", ".example"]);
const SKIP_DIRS = new Set(["node_modules", ".git", "dist", "build", "coverage", ".next", ".cache", "vendor", "__pycache__", ".venv"]);
const MAX_FILE_BYTES = 256 * 1024;

export function deriveKeywords(incident) {
  const keywords = new Set();
  for (const segment of incident.path.split("/")) {
    const clean = segment.replace(/[^a-zA-Z0-9_-]/g, "").toLowerCase();
    if (clean.length >= 3 && !/^\d+$/.test(clean)) keywords.add(clean);
  }
  const evidence = `${incident.responseSample ?? ""} ${incident.error?.message ?? ""}`.toLowerCase();
  for (const term of ["database", "pool", "timeout", "connection", "auth", "token", "econnrefused", "null", "undefined", "query"]) {
    if (evidence.includes(term)) keywords.add(term);
  }
  return [...keywords].slice(0, 12);
}

async function walk(root, maxFiles) {
  const files = [];
  const stack = [root];
  while (stack.length && files.length < maxFiles) {
    const dir = stack.pop();
    let entries;
    try { entries = await fs.readdir(dir, { withFileTypes: true }); } catch { continue; }
    for (const entry of entries) {
      if (files.length >= maxFiles) break;
      const full = path.join(dir, entry.name);
      if (entry.isDirectory()) { if (!SKIP_DIRS.has(entry.name)) stack.push(full); }
      else if (entry.isFile()) {
        const ext = path.extname(entry.name).toLowerCase();
        if (SOURCE_EXTENSIONS.has(ext) || entry.name === ".env.example") files.push(full);
      }
    }
  }
  return files;
}

function scoreContent(content, keywords) {
  const lower = content.toLowerCase();
  const lines = lower.split("\n");
  let bestLine = -1, bestLineScore = 0, score = 0;
  const matched = new Set();
  for (const kw of keywords) if (lower.includes(kw)) { matched.add(kw); score += 1; }
  for (let i = 0; i < lines.length; i++) {
    let lineScore = 0;
    for (const kw of keywords) if (lines[i].includes(kw)) lineScore += 1;
    if (lineScore > bestLineScore) { bestLineScore = lineScore; bestLine = i; }
  }
  return { score: score + bestLineScore, line: bestLine, matched: [...matched] };
}

function snippetAround(content, lineIndex, context = 2) {
  const lines = content.split("\n");
  const start = Math.max(0, lineIndex - context);
  const end = Math.min(lines.length, lineIndex + context + 1);
  return lines.slice(start, end).map((l, i) => `${start + i + 1}: ${l}`).join("\n").slice(0, 1200);
}

export async function inspectCodebase(incident, opts = {}) {
  const root = opts.root ?? config.CODEBASE_PATH;
  const maxFiles = opts.maxFiles ?? config.CODEBASE_MAX_FILES;
  const maxFindings = opts.maxFindings ?? config.CODEBASE_MAX_FINDINGS;
  if (!root) return [];

  const resolvedRoot = path.resolve(root);
  try { if (!(await fs.stat(resolvedRoot)).isDirectory()) return []; } catch { return []; }

  const keywords = deriveKeywords(incident);
  if (!keywords.length) return [];

  const files = await walk(resolvedRoot, maxFiles);
  const scored = [];
  for (const file of files) {
    try {
      if ((await fs.stat(file)).size > MAX_FILE_BYTES) continue;
      const content = await fs.readFile(file, "utf8");
      const { score, line, matched } = scoreContent(content, keywords);
      if (score <= 0 || line < 0) continue;
      const relative = path.relative(resolvedRoot, file);
      scored.push({ score, finding: { file: relative, line: line + 1, snippet: snippetAround(content, line), reason: `Matched incident keywords: ${matched.join(", ")}` } });
    } catch { /* skip unreadable */ }
  }
  scored.sort((a, b) => b.score - a.score);
  return scored.slice(0, maxFindings).map(s => s.finding);
}

/** Read-only client for the Agent Connector. Outages become error results, never throws. */
export function makeConnectorClient(cfg = defaultConfig) {
  return async function investigateDependencies(services) {
    const controller = new AbortController();
    const timeout = setTimeout(() => controller.abort(), cfg.REQUEST_TIMEOUT_MS);
    try {
      const res = await fetch(new URL("/connector/investigate", cfg.CONNECTOR_URL), {
        method: "POST",
        headers: { "Content-Type": "application/json", Authorization: `Bearer ${cfg.CONNECTOR_API_TOKEN}` },

      });

export function createInvestigator(deps) {
  const { store, investigateDependencies, inspectCodebase, reason, notifyEngineer, config, services } = deps;

  function decideEscalation(incident, result, findings, connectorError, recurring) {
    const reasons = [];
    if (result.confidence < config.ESCALATION_CONFIDENCE_THRESHOLD) reasons.push(`low confidence (${(result.confidence * 100).toFixed(0)}%)`);
    if (result.severity === "critical") reasons.push("critical severity");
    if (connectorError) reasons.push("connector unreachable");
    if (hasUnreachableDependency(findings)) reasons.push("a dependency is unreachable");
    if (incident.category === "upstream_unreachable" || incident.category === "timeout") reasons.push("application server unreachable");
    if (recurring >= 3) reasons.push(`recurring failure (${recurring + 1}x on this path)`);
    return { escalate: reasons.length > 0, escalationReason: reasons.join("; ") || undefined };
  }

  async function process(record) {
    const incident = record.incident;
    const recurring = store.recurrenceForPath(incident.path, incident.id);
    log("info", "investigation_started", { incidentId: incident.id, path: incident.path, recurring });

    const [dependency, codeFindings] = await Promise.all([investigateDependencies(services), inspectCodebase(incident)]);
    const recentCommits = config.codebase?.commits || [];
    const repoUrl = config.codebase?.repoUrl;
    const result = await reason({ incident, dependency, codeFindings, recurring, recentCommits, repoUrl });
    const dependencyFindings = deriveDependencyFindings(dependency);
    const { escalate, escalationReason } = decideEscalation(incident, result, dependencyFindings, dependency.error, recurring);

    const analysis = {
      incidentId: incident.id, mode: result.mode, severity: result.severity, summary: result.summary,
      rootCause: result.rootCause, suggestedFixes: result.suggestedFixes, dependencyFindings, codeFindings,
      confidence: result.confidence, escalate, escalationReason, reasoning: result.reasoning, recurring,
      recentCommits, repoUrl,
      policy: { dryRun: config.DRY_RUN, autoRollback: config.AUTO_ROLLBACK, escalationThreshold: config.ESCALATION_CONFIDENCE_THRESHOLD },
      monitoringActive: config.MONITORING_ACTIVE,
      createdAt: new Date().toISOString()
    };
    store.attachAnalysis(incident.id, analysis);
    log("info", "incident_analyzed", { incidentId: incident.id, mode: analysis.mode, severity: analysis.severity, confidence: analysis.confidence, escalate });

    if (escalate) {
      const notification = await notifyEngineer(incident, analysis, escalationReason || "escalation policy");
      store.setNotification(incident.id, notification);
    }
    return store.get(incident.id);
  }

  function startInvestigation(incident) {
    const record = store.create(incident);
    const completion = process(record).catch(e => {
      log("error", "investigation_failed", { incidentId: incident.id, error: e.message });
      return store.get(incident.id);
    });
    return { record, completion };
  }

  return { startInvestigation, runInvestigation: incident => startInvestigation(incident).completion };
}

// Reasoning: dependency findings + deterministic heuristic + optional Groq LLM.
import { config as defaultConfig } from "./config.mjs";

const STATUS_RANK = { healthy: 0, unknown: 1, degraded: 2, unhealthy: 3, unreachable: 4 };
const worst = (a, b) => (STATUS_RANK[a] >= STATUS_RANK[b] ? a : b);

function signalsFromEvidence(e) {
  const signals = [];
  const data = e.data || {};
  if (e.error) signals.push(`${e.check} check failed: ${e.error.code} ${e.error.message}`);
  if (e.check === "health" && data.checks) for (const [name, status] of Object.entries(data.checks)) if (status !== "healthy") signals.push(`${name}: ${status}`);
  if (e.check === "metrics" && data.metrics) {
    const m = data.metrics;
    if (m.connection_pool_usage_percent >= 90) signals.push(`connection pool ${m.connection_pool_usage_percent}% used`);
    if (m.error_rate_percent > 5) signals.push(`error rate ${m.error_rate_percent}%`);
    if (m.average_latency_ms >= 1000) signals.push(`avg latency ${m.average_latency_ms}ms`);
    if (m.average_query_ms >= 1000) signals.push(`avg query ${m.average_query_ms}ms`);
    if (m.http_status_code >= 500) signals.push(`upstream HTTP ${m.http_status_code}`);
    if (m.timeout_count_last_5m > 0) signals.push(`${m.timeout_count_last_5m} timeouts in 5m`);
  }
  if (e.check === "logs" && Array.isArray(data.logs)) for (const log of data.logs.filter(l => l.level === "error").slice(0, 2)) signals.push(`log: ${log.message}`);
  return signals;
}

export function deriveDependencyFindings(investigation) {
  const byService = new Map();
  for (const e of investigation.results) {
    const existing = byService.get(e.service) || { service: e.service, status: "healthy", signals: [] };
    existing.status = worst(existing.status, e.status);
    existing.signals.push(...signalsFromEvidence(e));
    byService.set(e.service, existing);
  }
  return [...byService.values()].map(f => ({ ...f, signals: [...new Set(f.signals)].slice(0, 8) }));
}

export const hasUnreachableDependency = findings => findings.some(f => f.status === "unreachable");
export const unhealthyServices = findings => findings.filter(f => ["unhealthy", "unreachable", "degraded"].includes(f.status));

const fix = (title, description, kind, file) => ({ title, description, kind, file });

export function heuristicReason(input) {
  const { incident, dependency, codeFindings, recurring, recentCommits = [], repoUrl } = input;
  const findings = deriveDependencyFindings(dependency);
  const primary = unhealthyServices(findings)[0];
  let severity = "medium", summary, rootCause, confidence, fixes;
  const narrative = [`Incident: ${incident.method} ${incident.path} -> HTTP ${incident.statusCode} (${incident.category}).`];
  if (dependency.error) narrative.push(`Agent Connector was not fully reachable: ${dependency.error}.`);

  if (primary) {
    const svc = primary.service;
    narrative.push(`Dependency ${svc} is ${primary.status}. Signals: ${primary.signals.join("; ") || "none"}.`);
    if (svc === "database") {
      severity = "high";
      summary = `Request failed because the database dependency is ${primary.status}.`;
      rootCause = "Database degradation (connection-pool exhaustion, slow queries or connectivity) is causing the application request to fail.";
      confidence = primary.status === "unhealthy" ? 0.82 : 0.65;
      fixes = [
        fix("Verify database connectivity and pool", "Check the database is reachable and the connection pool is not exhausted; raise pool size or fix leaked connections.", "dependency"),
        fix("Add query timeouts and retry/backoff", "Wrap database calls with bounded timeouts and exponential backoff so transient failures degrade gracefully.", "code", codeFindings[0]?.file),
        fix("Inspect slow queries", "Review the slowest queries and add indexes or pagination where average query time is high.", "dependency")
      ];
    } else if (svc === "external_api") {
      severity = "high";
      summary = `Request failed because a third-party API dependency is ${primary.status}.`;
      rootCause = "An upstream third-party API is returning errors or timing out, propagating failure to this request.";
      confidence = primary.status === "unhealthy" ? 0.78 : 0.6;
      fixes = [
        fix("Add a circuit breaker and fallback", "Stop hammering a failing upstream; serve a cached/fallback response while it recovers.", "code", codeFindings[0]?.file),
        fix("Tune client timeouts and retries", "Set explicit timeouts and bounded retries with jitter on the third-party client.", "config"),
        fix("Confirm provider status", "Check the provider's status page and your API credentials/quota.", "investigation")
      ];
    } else {
      severity = "high";
      summary = `Request failed because the authentication dependency is ${primary.status}.`;
      rootCause = "Authentication service degradation (token signing or identity-store access) is causing failures.";
      confidence = primary.status === "unhealthy" ? 0.72 : 0.58;
      fixes = [
        fix("Verify auth service health", "Check token-signing keys and identity-store connectivity on the auth service.", "dependency"),
        fix("Handle auth failures gracefully", "Return a clear 401/503 and avoid cascading 500s when auth is degraded.", "code", codeFindings[0]?.file)
      ];
    }
  } else if (incident.category === "upstream_unreachable" || incident.category === "timeout") {
    severity = "high";
    summary = "The application Server itself was unreachable or timed out.";
    rootCause = "The Server process is down, crashed, or overloaded (no dependency failure was detected).";
    confidence = 0.7;
    fixes = [fix("Check Server process and health", "Confirm the Server is running and its /health endpoint responds; inspect restart/crash loops.", "investigation"), fix("Review resource saturation", "Check CPU/memory and consider scaling or a readiness probe.", "config")];
    narrative.push("No dependency failure detected; the Server endpoint did not respond.");
  } else if (incident.category === "client_error") {
    severity = "low";
    summary = `Client error HTTP ${incident.statusCode} on ${incident.path}.`;
    rootCause = "The requested route or resource does not exist or the request was malformed (e.g. missing route registration or wrong path).";
    confidence = 0.6;
    fixes = [fix("Verify route registration", `Confirm ${incident.path} is registered and the HTTP method matches.`, "code", codeFindings[0]?.file), fix("Check client request", "Validate the client is calling the correct path and parameters.", "investigation")];
    narrative.push("No dependency failure detected; this is a client-side/routing error.");
  } else if (incident.category === "slow_response") {
    severity = "medium";
    summary = `Slow response on ${incident.path} (${Math.round(incident.durationMs)}ms).`;
    rootCause = "The request exceeded the latency threshold — likely a performance regression, an unindexed query, or downstream latency.";
    confidence = 0.5;
    fixes = [fix("Profile the slow endpoint", "Trace the request to find the slow span (query, external call, or CPU).", "investigation", codeFindings[0]?.file), fix("Add caching / pagination / indexes", "Cache hot reads, paginate large results, and index slow queries.", "code", codeFindings[0]?.file)];
    narrative.push("No dependency outage detected; the response was slower than the configured threshold.");
  } else if (codeFindings.length > 0) {
    severity = "medium";
    summary = `Application error on ${incident.path}; relevant code located.`;
    rootCause = `An unhandled error in the application code path (see ${codeFindings[0].file}) is the most likely cause.`;
    confidence = 0.55;
    fixes = [fix("Add error handling around the failing path", "Inspect the matched handler and add validation / try-catch with a meaningful error response.", "code", codeFindings[0].file), fix("Reproduce with the captured request", "Use the incident request and response sample to reproduce locally.", "investigation")];
    narrative.push(`Dependencies look healthy; matched ${codeFindings.length} relevant code location(s).`);
  } else {
    severity = "medium";
    summary = `Unclassified HTTP ${incident.statusCode} on ${incident.path}.`;
    rootCause = "Insufficient evidence to determine a root cause automatically.";
    confidence = 0.3;
    fixes = [fix("Gather more evidence", "Enable richer application logs/tracing for this route and re-run the request.", "investigation"), fix("Manual inspection", "An engineer should review the request path and recent deployments.", "investigation")];
    narrative.push("No dependency failure and no relevant code found; evidence is insufficient.");
  }

  // Correlate the failing request with the last commits of the linked repository.
  if (recentCommits.length) {
    const pathWords = incident.path.split("/").map(s => s.replace(/[^a-zA-Z0-9_-]/g, "").toLowerCase()).filter(s => s.length >= 3 && !/^\d+$/.test(s));
    const correlated = recentCommits.find(c => pathWords.some(w => (c.message || "").toLowerCase().includes(w)));
    narrative.push(`Linked repo ${repoUrl || ""} — last ${recentCommits.length} commits: ${recentCommits.map(c => `${(c.sha || "").slice(0, 7)} ${(c.message || "").split("\n")[0]}`).join(" | ")}.`);
    if (correlated) {
      narrative.push(`Recent commit ${(correlated.sha || "").slice(0, 7)} touches this area ("${(correlated.message || "").split("\n")[0]}") — a likely regression source.`);
      fixes.unshift(fix(`Review recent commit ${(correlated.sha || "").slice(0, 7)}`, `This commit mentions the failing area: "${(correlated.message || "").split("\n")[0]}". Inspect its diff first.`, "code", correlated.url));
    }
  }

  if (recurring >= 3) {
    narrative.push(`This path has failed ${recurring + 1} times recently — recurring issue.`);
    if (severity === "low") severity = "medium"; else if (severity === "medium") severity = "high";
  }

  return { mode: "mock", severity, summary, rootCause: { summary: rootCause, confidence }, suggestedFixes: fixes, confidence, reasoning: narrative.join(" ") };
}

const SYSTEM_PROMPT = [

  "Respond with ONLY a JSON object (no markdown) with keys:",

export async function groqReason(input, cfg = defaultConfig) {
  if (!cfg.GROQ_API_KEY) return null;
  const evidence = {
    incident: { method: input.incident.method, path: input.incident.path, statusCode: input.incident.statusCode, category: input.incident.category, reason: input.incident.reason, responseSample: input.incident.responseSample?.slice(0, 800), error: input.incident.error },
    dependencies: deriveDependencyFindings(input.dependency),
    connectorError: input.dependency.error,
    codeFindings: input.codeFindings.map(f => ({ file: f.file, line: f.line, snippet: f.snippet.slice(0, 600), reason: f.reason })),
    recurringFailuresOnPath: input.recurring,
    repository: input.repoUrl,
    recentCommits: (input.recentCommits || []).map(c => ({ sha: (c.sha || "").slice(0, 7), message: (c.message || "").split("\n")[0], author: c.author, date: c.date }))
  };
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), cfg.LLM_TIMEOUT_MS);
  try {
    const res = await fetch(`${cfg.GROQ_BASE_URL.replace(/\/$/, "")}/chat/completions`, {
      method: "POST",
      headers: { "Content-Type": "application/json", Authorization: `Bearer ${cfg.GROQ_API_KEY}` },
      signal: controller.signal,
      body: JSON.stringify({ model: cfg.GROQ_MODEL, temperature: 0.2, max_tokens: 1200, response_format: { type: "json_object" }, messages: [{ role: "system", content: SYSTEM_PROMPT }, { role: "user", content: `Diagnose this incident and return the JSON object.\n\nEVIDENCE:\n${JSON.stringify(evidence)}` }] })
    });
    if (!res.ok) return null;
    const data = await res.json();
    const content = data.choices?.[0]?.message?.content;
    if (!content) return null;
    const v = JSON.parse(content);
    if (!v.severity || !v.rootCause || typeof v.confidence !== "number" || !Array.isArray(v.suggestedFixes)) return null;
    return { mode: "live", severity: v.severity, summary: String(v.summary || ""), rootCause: { summary: String(v.rootCause.summary || ""), confidence: Number(v.rootCause.confidence) || 0 }, suggestedFixes: v.suggestedFixes.slice(0, 8), confidence: Number(v.confidence), reasoning: String(v.reasoning || "") };
  } catch {
    return null;
  } finally {
    clearTimeout(timeout);
  }
}

export async function reason(input, cfg = defaultConfig) {
  if (cfg.AGENT_MODE === "live" && cfg.GROQ_API_KEY) {
    const live = await groqReason(input, cfg);
    if (live) return live;
  }
  return heuristicReason(input);
}

function readJson(req, limit = 65536) {
  return new Promise((resolve, reject) => {
    let size = 0; const chunks = [];
    req.on("data", c => { size += c.length; if (size > limit) { reject(new Error("body too large")); req.destroy(); } else chunks.push(c); });
    req.on("end", () => { try { resolve(chunks.length ? JSON.parse(Buffer.concat(chunks).toString("utf8")) : {}); } catch { reject(new Error("invalid JSON")); } });
    req.on("error", reject);
  });
}

    codebase: { repoUrl: config.codebase.repoUrl || null, fullName: config.codebase.fullName || null, commitCount: config.codebase.commits.length }
  };
}

function applyConfig(body) {
  if (typeof body.escalationThreshold === "number" && body.escalationThreshold >= 0 && body.escalationThreshold <= 1) config.ESCALATION_CONFIDENCE_THRESHOLD = body.escalationThreshold;
  if (typeof body.dryRun === "boolean") config.DRY_RUN = body.dryRun;
  if (typeof body.autoRollback === "boolean") config.AUTO_ROLLBACK = body.autoRollback;
  if (typeof body.monitoringActive === "boolean") config.MONITORING_ACTIVE = body.monitoringActive;
  if (body.codebase && typeof body.codebase === "object") {
    if (typeof body.codebase.repoUrl === "string") config.codebase.repoUrl = body.codebase.repoUrl || undefined;
    if (typeof body.codebase.fullName === "string") config.codebase.fullName = body.codebase.fullName || undefined;
    if (Array.isArray(body.codebase.commits)) config.codebase.commits = body.codebase.commits.slice(0, 5);
  }
  log("info", "agent_core_config_updated", publicConfig());
}

function validateIncident(b) {
  if (!b || typeof b !== "object") return "incident must be an object";
  for (const f of ["id", "requestId", "method", "path", "category", "reason", "detectedAt", "upstream"]) {
    if (typeof b[f] !== "string" || !b[f]) return `missing field: ${f}`;
  }
  if (!Number.isInteger(b.statusCode)) return "statusCode must be an integer";
  if (typeof b.durationMs !== "number") return "durationMs must be a number";
  return null;
}

const server = http.createServer(async (req, res) => {
  const url = new URL(req.url, "http://localhost");
  const path = url.pathname;
  const incoming = req.headers["x-request-id"];
  const requestId = incoming && /^[a-zA-Z0-9._:-]{1,100}$/.test(incoming) ? incoming : randomUUID();
  res.setHeader("X-Request-Id", requestId);
  const start = process.hrtime.bigint();
  res.on("finish", () => log(res.statusCode >= 500 ? "error" : "info", "http_request_completed", { requestId, method: req.method, path, statusCode: res.statusCode, durationMs: Math.round(Number(process.hrtime.bigint() - start) / 1e6 * 100) / 100 }));

  try {
    if (path === "/health" && req.method === "GET") return sendJson(res, 200, { success: true, service: "agent-core", status: "healthy", mode: config.AGENT_MODE, monitoringActive: config.MONITORING_ACTIVE });

    // Runtime configuration from the control plane (onboarding Steps 03 + 04).
    if (path === "/config") {
      if (!authed(req)) return err(res, 401, "UNAUTHORIZED", "Valid agent-core bearer token required");
      if (req.method === "GET") return sendJson(res, 200, { success: true, config: publicConfig() });
      if (req.method === "POST") { applyConfig(await readJson(req)); return sendJson(res, 200, { success: true, config: publicConfig() }); }
    }

    if (!path.startsWith("/incidents")) return err(res, 404, "NOT_FOUND", "Route not found");
    if (!authed(req)) return err(res, 401, "UNAUTHORIZED", "Valid agent-core bearer token required");

    if (path === "/incidents" && req.method === "POST") {
      const body = await readJson(req);
      const problem = validateIncident(body);
      if (problem) return err(res, 400, "VALIDATION_ERROR", problem);
      // Monitoring switch (onboarding "Start monitoring"): when paused, record but don't investigate.
      if (!config.MONITORING_ACTIVE) {
        const record = store.create(body);
        record.status = "received";
        store.persist();
        return sendJson(res, 202, { success: true, incident: { id: record.incident.id, status: "received" }, monitoringActive: false });
      }
      const { record, completion } = investigator.startInvestigation(body);
      if (url.searchParams.get("wait") === "true") return sendJson(res, 200, { success: true, incident: await completion });
      completion.catch(() => {});
      return sendJson(res, 202, { success: true, incident: { id: record.incident.id, status: record.status }, statusUrl: `/incidents/${record.incident.id}` });
    }

    if (path === "/incidents" && req.method === "GET") {
      const limit = Math.min(Number(url.searchParams.get("limit")) || 50, config.MAX_CONTEXT_INCIDENTS);
      return sendJson(res, 200, { success: true, incidents: store.list(limit), context: store.contextSummary() });
    }

    if (path === "/incidents/context" && req.method === "GET") return sendJson(res, 200, { success: true, context: store.contextSummary() });

    const m = path.match(/^\/incidents\/([^/]+)(\/feedback)?$/);
    if (m) {
      const id = decodeURIComponent(m[1]);
      const isFeedback = Boolean(m[2]);

      if (!isFeedback && req.method === "GET") {
        const record = store.get(id);
        return record ? sendJson(res, 200, { success: true, incident: record }) : err(res, 404, "NOT_FOUND", "Incident not found");
      }

      if (isFeedback && req.method === "POST") {
        const body = await readJson(req);
        if (!["resolved", "unresolved"].includes(body.status)) return err(res, 400, "VALIDATION_ERROR", "status must be resolved or unresolved");
        const existing = store.get(id);
        if (!existing) return err(res, 404, "NOT_FOUND", "Incident not found");
        const record = store.setFeedback(id, body.status, body.note);
        if (body.status === "unresolved" && record.analysis) {
          const notification = await notifyEngineer(record.incident, record.analysis, `engineer marked unresolved${body.note ? `: ${body.note}` : ""}`);
          store.setNotification(id, notification);
          log("warn", "incident_escalated_by_feedback", { incidentId: id });
        }
        return sendJson(res, 200, { success: true, incident: store.get(id) });
      }
    }

    return err(res, 404, "NOT_FOUND", "Route not found");
  } catch (e) {
    return err(res, 400, "BAD_REQUEST", e.message);
  }
});

server.listen(config.PORT, () => log("info", "agent_core_listening", { port: config.PORT, mode: config.AGENT_MODE, connector: config.CONNECTOR_URL, codebase: config.CODEBASE_PATH || "(none)", llm: config.AGENT_MODE === "live" && config.GROQ_API_KEY ? config.GROQ_MODEL : "heuristic" }));

// Context retention: bounded in-memory history with optional disk persistence.
import fs from "node:fs";
import path from "node:path";

export class IncidentStore {
  constructor(max = 200, dataDir = undefined) {
    this.max = max;
    this.dataDir = dataDir;
    this.records = new Map();
    this.order = [];
  }

  file() { return this.dataDir ? path.join(this.dataDir, "incidents.json") : undefined; }

  load() {
    const file = this.file();
    if (!file) return;
    try {
      const arr = JSON.parse(fs.readFileSync(file, "utf8"));
      for (const record of arr.slice(-this.max)) { this.records.set(record.incident.id, record); this.order.push(record.incident.id); }
    } catch { /* no prior context */ }
  }

  persist() {
    const file = this.file();
    if (!file) return;
    try {
      fs.mkdirSync(this.dataDir, { recursive: true });
      fs.writeFileSync(file, JSON.stringify(this.order.map(id => this.records.get(id)).filter(Boolean), null, 2));
    } catch { /* best-effort */ }
  }

  create(incident) {
    const now = new Date().toISOString();
    const record = { incident, status: "investigating", createdAt: now, updatedAt: now };
    this.records.set(incident.id, record);
    this.order.push(incident.id);
    while (this.order.length > this.max) this.records.delete(this.order.shift());
    this.persist();
    return record;
  }

  attachAnalysis(id, analysis) {
    const r = this.records.get(id);
    if (!r) return undefined;
    r.analysis = analysis; r.status = "analyzed"; r.updatedAt = new Date().toISOString();
    this.persist();
    return r;
  }

  setNotification(id, notification) {
    const r = this.records.get(id);
    if (!r) return;
    r.notification = notification; r.updatedAt = new Date().toISOString();
    this.persist();
  }

  setFeedback(id, status, note) {
    const r = this.records.get(id);
    if (!r) return undefined;
    r.feedback = { status, note, at: new Date().toISOString() };
    r.status = status; r.updatedAt = new Date().toISOString();
    this.persist();
    return r;
  }

  get(id) { return this.records.get(id); }
  list(limit = 50) { return this.order.slice(-limit).reverse().map(id => this.records.get(id)).filter(Boolean); }

  recurrenceForPath(p, excludeId) {
    let count = 0;
    for (const r of this.records.values()) if (r.incident.path === p && r.incident.id !== excludeId) count++;
    return count;
  }

  contextSummary() {
    const records = [...this.records.values()];
    const byStatus = {}, byPath = {};
    for (const r of records) {
      byStatus[r.status] = (byStatus[r.status] || 0) + 1;
      byPath[r.incident.path] = (byPath[r.incident.path] || 0) + 1;
    }
    const recurring = Object.entries(byPath).filter(([, n]) => n >= 3).sort((a, b) => b[1] - a[1]).map(([p, n]) => ({ path: p, count: n }));
    return { total: records.length, byStatus, recurring, escalated: records.filter(r => r.analysis?.escalate).length };
  }
}

// Real network checks used by the onboarding UI. No mocked setTimeouts.
import dns from "node:dns/promises";
import { config } from "./config.mjs";

async function fetchWithTimeout(url, options, timeoutMs) {
  const controller = new AbortController();
  const t = setTimeout(() => controller.abort(), timeoutMs);
  try { return await fetch(url, { ...options, signal: controller.signal }); }
  finally { clearTimeout(t); }
}

/**
 * Real reachability check.
 * - http(s): performs an actual request; ANY HTTP response (even 401/404) means reachable.
 * - other schemes (mongodb+srv, postgres, redis…): resolves DNS for the host. A full
 *   protocol handshake would require a driver, so this is a DNS-level reachability probe.
 */
export async function checkReachability(rawUrl, { token, timeoutMs = config.REQUEST_TIMEOUT_MS } = {}) {
  const started = Date.now();
  let url;
  try { url = new URL(rawUrl); } catch { return { ok: false, reachable: false, error: "Invalid URL" }; }

  if (url.protocol === "http:" || url.protocol === "https:") {
    try {
      const res = await fetchWithTimeout(url.toString(), { method: "GET", headers: token ? { Authorization: `Bearer ${token}` } : {} }, timeoutMs);
      return { ok: true, reachable: true, method: "http", status: res.status, latencyMs: Date.now() - started };
    } catch (e) {
      return { ok: false, reachable: false, method: "http", error: e.name === "AbortError" ? "Timed out" : (e.cause?.code || e.message || "Unreachable"), latencyMs: Date.now() - started };
    }
  }

  // Non-HTTP: DNS-level probe.
  try {
    const host = url.hostname;
    if (!host) return { ok: false, reachable: false, error: "No host in URL" };
    const records = await dns.lookup(host);
    return { ok: true, reachable: true, method: "dns", address: records.address, latencyMs: Date.now() - started, note: `${url.protocol} reachability verified via DNS (protocol handshake not performed)` };
  } catch (e) {
    return { ok: false, reachable: false, method: "dns", error: e.code || e.message || "DNS lookup failed", latencyMs: Date.now() - started };
  }
}

/** Parse owner/repo from a GitHub URL, git remote, or "owner/repo" shorthand. */
export function parseRepoUrl(input) {
  if (!input || typeof input !== "string") return null;
  const s = input.trim().replace(/\.git$/, "");
  let m = s.match(/github\.com[/:]([^/\s]+)\/([^/\s]+)$/i);
  if (m) return { owner: m[1], repo: m[2] };
  m = s.match(/^([^/\s:@]+)\/([^/\s:@]+)$/);
  if (m) return { owner: m[1], repo: m[2] };
  return null;
}

/** Fetch real repository metadata + the last 5 commits from the GitHub API. */
export async function fetchGitHubRepo(repoUrl, token, { timeoutMs = config.REQUEST_TIMEOUT_MS } = {}) {
  const parsed = parseRepoUrl(repoUrl);
  if (!parsed) throw new Error("Could not parse repository URL (expected https://github.com/owner/repo)");
  const { owner, repo } = parsed;
  const headers = { Accept: "application/vnd.github+json", "User-Agent": "sbwk-control-plane", "X-GitHub-Api-Version": "2022-11-28" };
  if (token) headers.Authorization = `Bearer ${token}`;
  const base = `${config.GITHUB_API_BASE}/repos/${owner}/${repo}`;

  const metaRes = await fetchWithTimeout(base, { headers }, timeoutMs);
  if (metaRes.status === 404) throw new Error("Repository not found, or the token lacks access");
  if (metaRes.status === 401) throw new Error("Invalid or expired access token");
  if (metaRes.status === 403) throw new Error("GitHub API forbidden or rate-limited");
  if (!metaRes.ok) throw new Error(`GitHub returned HTTP ${metaRes.status}`);
  const meta = await metaRes.json();

  const commitsRes = await fetchWithTimeout(`${base}/commits?per_page=5`, { headers }, timeoutMs);
  if (!commitsRes.ok) throw new Error(`Could not read commits (HTTP ${commitsRes.status})`);
  const rawCommits = await commitsRes.json();
  const commits = (Array.isArray(rawCommits) ? rawCommits : []).slice(0, 5).map(c => ({
    sha: c.sha,
    message: c.commit?.message || "",
    author: c.author?.login || c.commit?.author?.name || "unknown",
    date: c.commit?.author?.date || null,
    url: c.html_url || null
  }));

  return { fullName: meta.full_name, repoUrl: meta.html_url || repoUrl, defaultBranch: meta.default_branch, private: meta.private, description: meta.description || "", commits };
}

function readJson(req, limit = 65536) {
  return new Promise((resolve, reject) => {
    let size = 0; const chunks = [];
    req.on("data", c => { size += c.length; if (size > limit) { reject(new Error("body too large")); req.destroy(); } else chunks.push(c); });
    req.on("end", () => { try { resolve(chunks.length ? JSON.parse(Buffer.concat(chunks).toString("utf8")) : {}); } catch { reject(new Error("invalid JSON")); } });
    req.on("error", reject);
  });
}

export function classifyStatus(statusCode, minStatus) {
  if (statusCode >= 500) return { detect: statusCode >= minStatus, category: "server_error", reason: `Upstream returned server error HTTP ${statusCode}` };
  if (statusCode >= 400) return { detect: statusCode >= minStatus, category: "client_error", reason: `Upstream returned client error HTTP ${statusCode}` };
  return { detect: false, category: "server_error", reason: "Normal response" };
}

export function categoryForUpstreamError(code) {
  return code === "TIMEOUT" ? "timeout" : "upstream_unreachable";
}

export class DispatchThrottle {
  constructor(cooldownMs) { this.cooldownMs = cooldownMs; this.last = new Map(); }
  static key(i) { return `${i.method} ${i.path} ${i.statusCode}`; }
  shouldDispatch(i, now = Date.now()) {
    const key = DispatchThrottle.key(i);
    const prev = this.last.get(key);
    if (prev !== undefined && now - prev < this.cooldownMs) return false;
    this.last.set(key, now);
    if (this.last.size > 1000) for (const [k, t] of this.last) if (now - t >= this.cooldownMs) this.last.delete(k);
    return true;
  }
}

export function buildIncident(input) {
  const idx = input.url.indexOf("?");
  const path = idx === -1 ? input.url : input.url.slice(0, idx);
  const query = idx === -1 ? "" : input.url.slice(idx + 1);
  return {
    id: input.id,
    requestId: input.requestId,
    source: "middleware",
    detectedAt: new Date().toISOString(),
    method: input.method,
    path,
    query: query || undefined,
    statusCode: input.statusCode,
    durationMs: Math.round(input.durationMs * 100) / 100,
    upstream: input.upstream,
    category: input.category,
    reason: input.reason,
    responseSample: input.responseSample || undefined,
    clientIp: input.clientIp,
    userAgent: input.userAgent,
    error: input.error
  };
}

async function dispatchIncident(incident) {
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), config.DISPATCH_TIMEOUT_MS);
  try {

      headers: { "Content-Type": "application/json", Authorization: `Bearer ${config.AGENT_CORE_TOKEN}`, "X-Request-Id": incident.requestId },
      body: JSON.stringify(incident), signal: controller.signal

}

function sendJson(res, status, obj) { res.writeHead(status, { "Content-Type": "application/json" }); res.end(JSON.stringify(obj)); }

function adminAuthed(req) {
  if (config.NODE_ENV === "test") return true;
  const h = req.headers["authorization"] || "";
  const token = h.startsWith("Bearer ") ? h.slice(7) : "";
  const expected = config.MIDDLEWARE_ADMIN_TOKEN;
  return token.length === expected.length && timingSafeEqual(Buffer.from(token), Buffer.from(expected));
}

function clientIp(req) {
  const fwd = req.headers["x-forwarded-for"];
  if (typeof fwd === "string" && fwd.length) return fwd.split(",")[0].trim();
  return req.socket.remoteAddress;
}

function readJson(req, limit = 32768) {
  return new Promise((resolve, reject) => {
    let size = 0; const chunks = [];
    req.on("data", c => { size += c.length; if (size > limit) { reject(new Error("body too large")); req.destroy(); } else chunks.push(c); });
    req.on("end", () => { try { resolve(chunks.length ? JSON.parse(Buffer.concat(chunks).toString("utf8")) : {}); } catch { reject(new Error("invalid JSON")); } });
    req.on("error", reject);
  });
}

async function handleAdmin(req, res, url) {
  if (!adminAuthed(req)) return sendJson(res, 401, { success: false, error: { code: "UNAUTHORIZED", message: "Valid middleware admin bearer token required" } });

  if (url.pathname === "/_sbwk/status") {
    return sendJson(res, 200, { success: true, service: "middleware", target: config.TARGET_SERVER_URL, agentCore: config.AGENT_CORE_URL, detection: { minStatus: config.DETECT_STATUS_MIN, slowMs: config.DETECT_SLOW_MS, upstreamErrors: config.DETECT_UPSTREAM_ERRORS, cooldownMs: config.DISPATCH_COOLDOWN_MS }, incidents: storeCounts() });
  }
  if (url.pathname === "/_sbwk/incidents") {
    const limit = Math.min(Number(url.searchParams.get("limit")) || 50, config.MAX_RECENT_INCIDENTS);
    return sendJson(res, 200, { success: true, incidents: recent.slice(-limit).reverse() });
  }
  // Runtime configuration from the control plane (onboarding Step 01).
  if (url.pathname === "/_sbwk/config" && req.method === "POST") {
    const body = await readJson(req);
    if (typeof body.targetServerUrl === "string" && body.targetServerUrl) {
      try { new URL(body.targetServerUrl); config.TARGET_SERVER_URL = body.targetServerUrl; }
      catch { return sendJson(res, 400, { success: false, error: { code: "VALIDATION_ERROR", message: "targetServerUrl must be a valid URL" } }); }
    }
    if (Number.isFinite(Number(body.detectStatusMin))) config.DETECT_STATUS_MIN = Number(body.detectStatusMin);
    if (Number.isFinite(Number(body.detectSlowMs))) config.DETECT_SLOW_MS = Number(body.detectSlowMs);
    log("info", "middleware_config_updated", { target: config.TARGET_SERVER_URL, minStatus: config.DETECT_STATUS_MIN, slowMs: config.DETECT_SLOW_MS });
    return sendJson(res, 200, { success: true, target: config.TARGET_SERVER_URL, detection: { minStatus: config.DETECT_STATUS_MIN, slowMs: config.DETECT_SLOW_MS } });
  }
  return sendJson(res, 404, { success: false, error: { code: "NOT_FOUND", message: "Route not found" } });
}

const server = http.createServer((req, res) => {
  const url = new URL(req.url, "http://localhost");
  const incoming = req.headers["x-request-id"];
  const requestId = incoming && /^[a-zA-Z0-9._:-]{1,100}$/.test(incoming) ? incoming : randomUUID();
  res.setHeader("X-Request-Id", requestId);
  const start = process.hrtime.bigint();
  const elapsed = () => Number(process.hrtime.bigint() - start) / 1e6;

  if (url.pathname === "/health" && req.method === "GET") return sendJson(res, 200, { success: true, service: "middleware", status: "healthy" });
  if (url.pathname.startsWith("/_sbwk/")) return void handleAdmin(req, res, url).catch(e => sendJson(res, 400, { success: false, error: { code: "BAD_REQUEST", message: e.message } }));

  const target = resolveTarget();
  const headers = { ...req.headers, host: target.url.host, "x-forwarded-for": clientIp(req), "x-forwarded-proto": "http", "x-request-id": requestId };
  const proxyReq = target.lib.request({ protocol: target.url.protocol, hostname: target.url.hostname, port: target.port, path: req.url, method: req.method, headers }, proxyRes => {
    const status = proxyRes.statusCode || 0;
    const statusDecision = classifyStatus(status, config.DETECT_STATUS_MIN);
    res.writeHead(status, proxyRes.headers);
    proxyRes.pipe(res);

    const max = config.RESPONSE_SAMPLE_BYTES;
    const chunks = []; let size = 0;
    proxyRes.on("data", c => { if (size < max) { chunks.push(c); size += c.length; } });
    proxyRes.on("end", () => {
      const durationMs = elapsed();
      let decision = statusDecision;
      if (!decision.detect && config.DETECT_SLOW_MS > 0 && durationMs > config.DETECT_SLOW_MS) {
        decision = { detect: true, category: "slow_response", reason: `Response took ${Math.round(durationMs)}ms (threshold ${config.DETECT_SLOW_MS}ms)` };
      }
      log("info", "http_request_proxied", { requestId, method: req.method, path: req.url, statusCode: status, durationMs: Math.round(durationMs * 100) / 100, anomaly: decision.detect });
      if (decision.detect) {
        const sample = max > 0 && decision.category !== "slow_response" ? Buffer.concat(chunks).subarray(0, max).toString("utf8") : undefined;
        recordAndDispatch(buildIncident({
          id: randomUUID(), requestId, method: req.method, url: req.url, statusCode: status, durationMs,
          upstream: config.TARGET_SERVER_URL, category: decision.category, reason: decision.reason,
          responseSample: sample, clientIp: clientIp(req), userAgent: req.headers["user-agent"]
        }));
      }
    });
  });

  proxyReq.setTimeout(config.PROXY_TIMEOUT_MS, () => { const e = new Error("Upstream request timed out"); e.code = "ETIMEDOUT"; proxyReq.destroy(e); });

  proxyReq.on("error", err => {
    const isTimeout = err.code === "ETIMEDOUT" || err.code === "ESOCKETTIMEDOUT" || /timeout/i.test(err.message);
    const status = isTimeout ? 504 : 502;
    log("error", "upstream_error", { requestId, method: req.method, path: req.url, code: err.code || "CONNECTION_FAILED", message: err.message });
    if (!res.headersSent) sendJson(res, status, { success: false, error: { code: isTimeout ? "GATEWAY_TIMEOUT" : "BAD_GATEWAY", message: "Upstream server is unavailable" } });
    else res.end();
    if (config.DETECT_UPSTREAM_ERRORS) {
      recordAndDispatch(buildIncident({
        id: randomUUID(), requestId, method: req.method, url: req.url, statusCode: status, durationMs: elapsed(),
        upstream: config.TARGET_SERVER_URL, category: categoryForUpstreamError(isTimeout ? "TIMEOUT" : "CONNECTION_FAILED"),
        reason: isTimeout ? "Upstream server timed out" : "Upstream server could not be reached",
        clientIp: clientIp(req), userAgent: req.headers["user-agent"], error: { code: err.code || "CONNECTION_FAILED", message: err.message }
      }));
    }
  });

  req.pipe(proxyReq);
});

server.listen(config.PORT, () => log("info", "middleware_listening", { port: config.PORT, target: config.TARGET_SERVER_URL, agentCore: config.AGENT_CORE_URL, detectStatusMin: config.DETECT_STATUS_MIN, detectSlowMs: config.DETECT_SLOW_MS }));

function shutdown(signal) { log("info", "middleware_shutdown", { signal }); server.close(() => process.exit(0)); setTimeout(() => process.exit(1), 10000).unref(); }
process.on("SIGTERM", () => shutdown("SIGTERM"));
process.on("SIGINT", () => shutdown("SIGINT"));

import { test } from "node:test";
import assert from "node:assert/strict";
import { classifyStatus, categoryForUpstreamError, buildIncident, DispatchThrottle } from "../detector.mjs";

test("flags 5xx at the default threshold", () => {
  const d = classifyStatus(500, 500);
  assert.equal(d.detect, true);
  assert.equal(d.category, "server_error");
});

test("does not flag 404 when threshold is 500", () => {
  assert.equal(classifyStatus(404, 500).detect, false);
});

test("flags 404 as client error when threshold is 400", () => {
  const d = classifyStatus(404, 400);
  assert.equal(d.detect, true);
  assert.equal(d.category, "client_error");
});

test("never flags 2xx/3xx", () => {
  assert.equal(classifyStatus(200, 400).detect, false);
  assert.equal(classifyStatus(302, 400).detect, false);
});

test("maps upstream errors to categories", () => {
  assert.equal(categoryForUpstreamError("TIMEOUT"), "timeout");
  assert.equal(categoryForUpstreamError("CONNECTION_FAILED"), "upstream_unreachable");
});

test("throttle suppresses duplicates within the window", () => {
  const t = new DispatchThrottle(10000);
  const base = { method: "GET", path: "/users", statusCode: 500 };
  assert.equal(t.shouldDispatch(base, 1000), true);
  assert.equal(t.shouldDispatch(base, 2000), false);
  assert.equal(t.shouldDispatch(base, 11001), true);
});

test("throttle treats different paths/statuses independently", () => {
  const t = new DispatchThrottle(10000);
  const base = { method: "GET", path: "/users", statusCode: 500 };
  assert.equal(t.shouldDispatch(base, 1000), true);
  assert.equal(t.shouldDispatch({ ...base, path: "/orders" }, 1000), true);
  assert.equal(t.shouldDispatch({ ...base, statusCode: 503 }, 1000), true);
});

test("buildIncident splits path/query and rounds duration", () => {

  assert.equal(i.path, "/users");
  assert.equal(i.query, "id=42");
  assert.equal(i.durationMs, 12.35);
  assert.equal(i.source, "middleware");
});

# SBWK (Sleepy Bot with Keys)

- I tell you where the problem is in the server, and suggest you fixes. I am all seeing eye.

- So essentially, we make a middleware, either self hosted or provided by us (future)

- Middleware is docker container??
- Now..


Ok, so this is what we do.

- A docker container, that takes inbound request and redirects request to server as normal. Like cloudflare.
- It also has access to other 3rd party servers, and databases that our servers use. Like auth server (google auth, mongodb auth); It acquires logs on the 3rd party servers too (the developer has to configure it.

- Noow then, A core agent is connected to our docker container.
- The docker container has a built in detector that detects abnormalities life 404, or 500 or anything.
- Therefore, once a abnormal request comes by, the dispatch is sent to core agent. MCP or something via API. And agent thinks and returns solution. THe agent first accesses external logs, finds errors. Then looks at the codebase. If error is found, it will suggest fix. 
- Agent retains context. If things don't work out, actual engineer is awoken.
